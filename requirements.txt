# All the Python packages of the course - for working with the whole repository (Python 3.12 or newer).
# A student needs only the requirements.txt of the archive of a module: it names the packages of that module.
#
#   python -m venv .venv
#   source .venv/bin/activate        (Windows: .venv\Scripts\activate)
#   pip install -r requirements.txt
#
# No versions: pip installs the newest versions that work with your Python.

fastapi            # web framework: pages and RESTful services
uvicorn            # web server that runs FastAPI applications
jinja2             # HTML templates
python-multipart   # reading the data of HTML forms
httpx2             # HTTP client: calling RESTful services from Python
sqlmodel           # working with a relational database through Python classes
supabase           # client for the Supabase cloud database
python-dotenv      # reading settings from the .env file
